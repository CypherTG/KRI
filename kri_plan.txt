1. Full System Architecture
1.1 The Three Layers (All M365)
Every component lives inside your Microsoft 365 tenant. There is no external infrastructure to provision, pay for, or maintain.


1.2 Data Flow (No External Calls)
User opens the KRI SharePoint page → SPFx web part loads inside SharePoint
Web part authenticates silently using SharePoint context — user is already logged in, no login screen appears
UI calls Microsoft Graph API directly using MSGraphClientV3 (built into every SPFx web part)
Graph API reads and writes your SharePoint Lists, sends emails via Outlook, posts Teams messages
Power Automate flows run on timers and SharePoint list triggers — no code, no servers

1.3 Why No Backend Server Is Needed
In a standard web app you need a backend server because your API keys must be kept secret. In SPFx inside SharePoint, authentication is handled entirely by your M365 tenant:
The user is already authenticated into SharePoint when they open the page
SPFx's MSGraphClientV3 automatically acquires a Graph API access token from SharePoint context — zero code required
That token only works for this user, in this tenant, with the permissions an admin has approved
There are no API keys to protect. There is no secret to store. There is no server to run.

5. Phase 2 — SPFx Entry Point & Configuration
5.1 KriPlatformWebPart.ts — SPFx Entry Point
This is the root of the web part. It acquires the MSGraphClientV3 — which handles all authentication automatically using the SharePoint page context — and passes it into your React application.

import * as React from 'react';
import * as ReactDom from 'react-dom';
import { Version } from '@microsoft/sp-core-library';
import { BaseClientSideWebPart } from '@microsoft/sp-webpart-base';
import { MSGraphClientV3 } from '@microsoft/sp-http';
import App from './components/App';

export interface IKriPlatformWebPartProps {}

export default class KriPlatformWebPart
  extends BaseClientSideWebPart<IKriPlatformWebPartProps> {

  private _graphClient: MSGraphClientV3;

  public async onInit(): Promise<void> {
    // SharePoint acquires the token silently — no MSAL code needed
    this._graphClient =
      await this.context.msGraphClientFactory.getClient('3');
    return super.onInit();
  }

  public render(): void {
    const element = React.createElement(App, {
      graphClient: this._graphClient,
      context: this.context,
    });
    ReactDom.render(element, this.domElement);
  }

  protected get dataVersion(): Version {
    return Version.parse('1.0');
  }
}

6.1 services/graphApi.ts — Complete Graph API Layer
import { MSGraphClientV3 } from '@microsoft/sp-http';

// Your SharePoint site ID — get it from:
// https://graph.microsoft.com/v1.0/sites/{domain}.sharepoint.com:/sites/{sitename}
const SITE_ID = 'YOUR_SHAREPOINT_SITE_ID';

// ── Clients ────────────────────────────────────────────────────────────
export async function getClients(graph: MSGraphClientV3) {
  const res = await graph
    .api(`/sites/${SITE_ID}/lists/KRI_Clients/items`)
    .expand('fields')
    .filter('fields/IsActive eq true')
    .get();
  return res.value.map((i: any) => i.fields);
}

export async function getClient(graph: MSGraphClientV3, id: string) {
  const res = await graph
    .api(`/sites/${SITE_ID}/lists/KRI_Clients/items/${id}`)
    .expand('fields').get();
  return res.fields;
}

export async function createClient(graph: MSGraphClientV3, data: any) {
  return graph
    .api(`/sites/${SITE_ID}/lists/KRI_Clients/items`)
    .post({ fields: {
      Title:                data.fullName,
      FirstName:            data.firstName,
      LastName:             data.lastName,
      CompanyName:          data.companyName,
      JobTitle:             data.jobTitle,
      Email:                data.email,
      Phone:                data.phone,
      Gender:               data.gender,
      KRIStage:             data.kriStage,
      ProjectName:          data.projectName,
      LinkedInProfile:      data.linkedIn,
      InstagramAccount:     data.instagram,
      KRISecondaryContact:  data.secondaryContact,
      BirthdayDate:         data.birthdayDate,
      IsActive:             true,
      DateRegistered:       new Date().toISOString(),
    }});
}

export async function updateClient(graph: MSGraphClientV3, id: string, data: any) {
  return graph
    .api(`/sites/${SITE_ID}/lists/KRI_Clients/items/${id}/fields`)
    .patch(data);
}

export async function updateStage(
  graph: MSGraphClientV3, id: string,
  newStage: string, previousStage: string, changedBy: string
) {
  // 1. Update the client's stage
  await graph
    .api(`/sites/${SITE_ID}/lists/KRI_Clients/items/${id}/fields`)
    .patch({ KRIStage: newStage });
  // 2. Log to StageHistory audit list
  await graph
    .api(`/sites/${SITE_ID}/lists/KRI_StageHistory/items`)
    .post({ fields: {
      Title:         `${id}: ${previousStage} -> ${newStage}`,
      ClientID:      id,
      PreviousStage: previousStage,
      NewStage:      newStage,
      ChangedDate:   new Date().toISOString(),
      ChangedBy:     changedBy,
    }});
}

export async function archiveClient(graph: MSGraphClientV3, id: string) {
  return graph
    .api(`/sites/${SITE_ID}/lists/KRI_Clients/items/${id}/fields`)
    .patch({ IsActive: false });
}

// ── Communications ─────────────────────────────────────────────────────
export async function getComms(graph: MSGraphClientV3, clientId: string) {
  const res = await graph
    .api(`/sites/${SITE_ID}/lists/KRI_Communications/items`)
    .expand('fields')
    .filter(`fields/ClientID eq '${clientId}'`)
    .orderby('fields/CommunicationDate desc')
    .get();
  return res.value.map((i: any) => i.fields);
}

export async function logComm(graph: MSGraphClientV3, data: any) {
  return graph
    .api(`/sites/${SITE_ID}/lists/KRI_Communications/items`)
    .post({ fields: {
      Title:              data.subject,
      ClientID:           data.clientId,
      CommunicationType:  data.type,
      Subject:            data.subject,
      Notes:              data.notes,
      CommunicationDate:  new Date().toISOString(),
      NextFollowUpDate:   data.nextFollowUp,
    }});
}

// ── Engagements ────────────────────────────────────────────────────────
export async function getEngagements(graph: MSGraphClientV3) {
  const res = await graph
    .api(`/sites/${SITE_ID}/lists/KRI_Engagements/items`)
    .expand('fields').get();
  return res.value.map((i: any) => i.fields);
}

export async function createEngagement(graph: MSGraphClientV3, data: any) {
  return graph
    .api(`/sites/${SITE_ID}/lists/KRI_Engagements/items`)
    .post({ fields: { ...data } });
}

// ── Send email via Outlook (Graph API — no SMTP, no server) ────────────
export async function sendEmail(
  graph: MSGraphClientV3,
  to: string, subject: string, body: string
) {
  return graph.api('/me/sendMail').post({
    message: {
      subject,
      body: { contentType: 'HTML', content: body },
      toRecipients: [{ emailAddress: { address: to } }],
    },
    saveToSentItems: true,
  });
}

7.2 components/clients/ClientList.tsx — Client Registry
import React, { useEffect, useState } from 'react';
import { Box, TextField, ToggleButton, ToggleButtonGroup,
         Grid, Typography, Fab, CircularProgress } from '@mui/material';
import AddIcon from '@mui/icons-material/Add';
import { MSGraphClientV3 } from '@microsoft/sp-http';
import { getClients } from '../../services/graphApi';
import ClientCard from './ClientCard';
import ClientForm from './ClientForm';

export default function ClientList({ graph }: { graph: MSGraphClientV3 }) {
  const [clients, setClients] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [search,  setSearch]  = useState('');
  const [stage,   setStage]   = useState('All');
  const [showForm, setShowForm] = useState(false);

  const load = () => {
    setLoading(true);
    getClients(graph).then(data => { setClients(data); setLoading(false); });
  };
  useEffect(load, []);

  const filtered = clients.filter(c => {
    const matchSearch = `${c.FirstName} ${c.LastName} ${c.CompanyName}`
      .toLowerCase().includes(search.toLowerCase());
    const matchStage = stage === 'All' || c.KRIStage === stage;
    return matchSearch && matchStage;
  });

  return (
    <Box sx={{ p: 3 }}>
      <Typography variant='h5' sx={{ mb: 2, fontWeight: 700 }}>
        Client Registry
      </Typography>
      <Box sx={{ display: 'flex', gap: 2, mb: 3 }}>
        <TextField size='small' label='Search clients...' value={search}
          onChange={e => setSearch(e.target.value)} sx={{ width: 280 }} />
        <ToggleButtonGroup exclusive value={stage}
          onChange={(_, v) => v && setStage(v)} size='small'>
          {['All','Knowledge','Rapport','Intimacy'].map(s => (
            <ToggleButton key={s} value={s}>{s}</ToggleButton>
          ))}
        </ToggleButtonGroup>
      </Box>
      {loading ? <CircularProgress /> : (
        <Grid container spacing={2}>
          {filtered.map((c, i) => (
            <Grid item xs={12} sm={6} md={4} key={i}>
              <ClientCard client={c} />
            </Grid>
          ))}
        </Grid>
      )}
      <Fab color='primary' sx={{ position: 'fixed', bottom: 32, right: 32 }}
        onClick={() => setShowForm(true)}>
        <AddIcon />
      </Fab>
      {showForm && <ClientForm graph={graph}
        onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); load(); }} />}
    </Box>
  );
}
